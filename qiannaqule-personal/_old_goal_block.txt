// ========== v52.8.9 目标设置（可设并回显） ==========
  function _goalKey(){ return (window.LS_PREFIX || 'mijieai_') + 'wealth_goal'; }
  window.getWealthGoal = function(){
    try { return JSON.parse(localStorage.getItem(_goalKey()) || 'null'); } catch(e) { return null; }
  };
  window.saveWealthGoal = function(){
    var t = document.getElementById('wg-type'); var a = document.getElementById('wg-amount'); var m = document.getElementById('wg-term');
    if (!t || !a) return;
    var goal = { type: t.value, amount: Math.round(parseFloat(a.value) || 0), term: Math.round(parseFloat(m && m.value) || 0) };
    try { localStorage.setItem(_goalKey(), JSON.stringify(goal)); } catch(e) {}
    try { var o = document.getElementById('goalSettingOverlay'); if (o) o.remove(); } catch(e) {}
    if (typeof initWealthGoalSummary === 'function') { try { initWealthGoalSummary(); } catch(e) {} }
  };
  window.initWealthGoalSummary = function(){
    var el = document.getElementById('wealthGoalSummary'); if (!el) return;
    var g = null; try { g = window.getWealthGoal(); } catch(e) {}
    if (g && g.type) {
      var tn = { emergency:'应急基金', saving:'月度储蓄', saving_target:'攒钱目标', repay:'还清贷款' };
      var text = '已设目标：'+ (tn[g.type]||g.type) + (g.amount>0?' · ¥'+g.amount.toLocaleString('zh-CN'):'') + (g.term>0?' · '+g.term+'个月':'');
      if (el.textContent !== text) el.textContent = text;
    } else {
      var d = '当前处于财务生存阶段。下一步目标：建立 3-6 个月应急基金，配置基础保障，进入财务安全阶段。';
      if (el.textContent !== d) el.textContent = d;
    }
  };
  window.openGoalSetting = function(){
    try { var od = document.getElementById('goalSettingOverlay'); if (od) od.remove(); } catch(e) {}
    var g = null; try { g = window.getWealthGoal(); } catch(e) {}
    var mask = document.createElement('div');
    mask.id = 'goalSettingOverlay';
    mask.style.cssText = 'position:fixed;top:0;left:0;right:0;bottom:0;background:rgba(0,0,0,0.5);z-index:12000;display:flex;align-items:center;justify-content:center;padding:20px';
    mask.onclick = function(e){ if (e.target === this) this.remove(); };
    var inner = document.createElement('div');
    inner.style.cssText = 'background:#fff;border-radius:16px;width:100%;max-width:360px;padding:20px;box-shadow:0 10px 40px rgba(0,0,0,0.2)';
    var opts = [['emergency','应急基金'],['saving','月度储蓄'],['saving_target','攒钱目标'],['repay','还清贷款']];
    var selOpts = opts.map(function(o){ return '<option value="'+o[0]+'"'+(g&&g.type===o[0]?' selected':'')+'>'+o[1]+'</option>'; }).join('');
    inner.innerHTML =
      '<div style="display:flex;align-items:center;justify-content:space-between;margin-bottom:14px">' +
        '<div style="font-size:17px;font-weight:700;color:#1f2937">设置财务目标</div>' +
        '<div style="cursor:pointer;color:#64748b;font-size:20px" onclick="(function(){var o=document.getElementById(\'goalSettingOverlay\');if(o)o.remove();})()">&times;</div>' +
      '</div>' +
      '<div style="font-size:13px;color:#64748b;margin-bottom:4px">目标类型</div>' +
      '<select id="wg-type" style="font-size:16px;width:100%;padding:9px;border:1px solid #e2e8f0;border-radius:10px;margin-bottom:12px">'+selOpts+'</select>' +
      '<div style="font-size:13px;color:#64748b;margin-bottom:4px">目标金额（元）</div>' +
      '<input type="number" id="wg-amount" value="'+(g&&g.amount?g.amount:'')+'" placeholder="如 50000" min="0" style="font-size:16px;width:100%;padding:9px;border:1px solid #e2e8f0;border-radius:10px;margin-bottom:12px;box-sizing:border-box">' +
      '<div style="font-size:13px;color:#64748b;margin-bottom:4px">目标期限（月，可留空）</div>' +
      '<input type="number" id="wg-term" value="'+(g&&g.term?g.term:'')+'" placeholder="如 12" min="0" style="font-size:16px;width:100%;padding:9px;border:1px solid #e2e8f0;border-radius:10px;margin-bottom:18px;box-sizing:border-box">' +
      '<button onclick="window.saveWealthGoal()" style="width:100%;padding:11px;font-size:15px;font-weight:600;color:#fff;background:#2563eb;border:none;border-radius:12px;cursor:pointer">保存目标</button>';
    mask.appendChild(inner);
    document.body.appendChild(mask);
  };
